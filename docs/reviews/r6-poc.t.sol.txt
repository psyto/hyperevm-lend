pragma solidity ^0.8.20;

import {Test} from "forge-std/Test.sol";
import {LendingMarket, IERC20} from "../../src/LendingMarket.sol";

contract Round6USDC {
    mapping(address => uint256) public balanceOf;
    mapping(address => mapping(address => uint256)) public allowance;

    function mint(address to, uint256 amount) external { balanceOf[to] += amount; }
    function approve(address spender, uint256 amount) external returns (bool) {
        allowance[msg.sender][spender] = amount;
        return true;
    }
    function transfer(address to, uint256 amount) external returns (bool) {
        balanceOf[msg.sender] -= amount;
        balanceOf[to] += amount;
        return true;
    }
    function transferFrom(address from, address to, uint256 amount) external returns (bool) {
        allowance[from][msg.sender] -= amount;
        balanceOf[from] -= amount;
        balanceOf[to] += amount;
        return true;
    }
}

/// Focused PoCs for the r5 payment-rounding change.
contract Round6PaymentMathPoC is Test {
    address internal constant MARK_PX = 0x0000000000000000000000000000000000000806;
    address internal constant ORACLE_PX = 0x0000000000000000000000000000000000000807;
    address internal constant BBO = 0x000000000000000000000000000000000000080e;
    uint32 internal constant HYPE_PERP = 135;

    Round6USDC internal usdc;
    LendingMarket internal market;
    address internal constant LENDER = address(0x601);
    address internal constant BORROWER = address(0x602);

    function setUp() public {
        usdc = new Round6USDC();
        market = new LendingMarket(IERC20(address(usdc)), 0, HYPE_PERP, 2, 100, 100, 100);
        _quote(40e4);
        usdc.mint(LENDER, 1_000e6);
        vm.startPrank(LENDER);
        usdc.approve(address(market), type(uint256).max);
        market.supply(1_000e6);
        vm.stopPrank();
    }

    function _quote(uint64 price4) internal {
        uint64 bid = price4 > 1 ? price4 - price4 / 2000 : 1;
        uint64 ask = price4 > 1 ? price4 + price4 / 2000 : 2;
        vm.mockCall(ORACLE_PX, abi.encode(HYPE_PERP), abi.encode(price4));
        vm.mockCall(MARK_PX, abi.encode(HYPE_PERP), abi.encode(price4));
        vm.mockCall(BBO, abi.encode(HYPE_PERP), abi.encode(bid, ask));
    }

    function _open(address borrower, uint256 coll, uint256 debt) internal {
        vm.deal(borrower, coll);
        vm.startPrank(borrower);
        market.depositCollateral{value: coll}();
        market.borrow(debt);
        usdc.approve(address(market), type(uint256).max);
        vm.stopPrank();
    }

    function test_exactFractionIsRoundedOnlyOnceAtTheEnd() public {
        // At $0.0001, this is exactly 1 USDC base unit of collateral.  Applying the
        // 5% discount gives 20/21 base units, which must round to one, not zero.
        _open(BORROWER, 0.01 ether, 1);
        _quote(1);

        uint256 cashBefore = usdc.balanceOf(BORROWER);
        vm.prank(BORROWER);
        uint256 seized = market.liquidate(BORROWER, 1);

        assertEq(cashBefore - usdc.balanceOf(BORROWER), 1);
        assertEq(seized, 0.01 ether);
        assertEq(market.debtShares(BORROWER), 0);
        emit log_named_uint("exact discounted value numerator / denominator", 20);
        emit log_named_uint("payment (USDC base units)", 1);
    }

    function test_roundingPremiumIsStrictlyLessThanOneBaseUnit() public {
        // At $0.0001 the collateral's exact value is 1.0500001 USDC base units.
        // Its discounted value is just over one unit, hence payment rounds to two.
        // The resulting excess over the undiscounted collateral value is 0.9499999
        // base units, proving the market-favouring rounding is bounded by one unit.
        uint256 coll = 10_500_001_000_000_000;
        _open(BORROWER, coll, 10);
        _quote(1);

        uint256 cashBefore = usdc.balanceOf(BORROWER);
        vm.prank(BORROWER);
        market.liquidate(BORROWER, type(uint256).max);
        uint256 paid = cashBefore - usdc.balanceOf(BORROWER);

        assertEq(paid, 2);
        assertGt(paid * 1e16, coll); // payment exceeds exact collateral value here
        assertLt(paid * 1e16 - coll, 1e16); // but by less than one USDC base unit
        assertEq(market.debtShares(BORROWER), 0);
        emit log_named_uint("payment premium, scaled by 1e16", paid * 1e16 - coll);
    }

    function test_POC_priceTimesBpsOverflowsForAcceptedLargeSzDecimals() public {
        // `szDecimals` is constructor-controlled and not range-checked.  At 62, a
        // raw price of one is 1e74 after HLPrecompiles' scaling.  The position is
        // validly opened at 60% LTV, then accrues into the short path.  The new
        // `price * BPS` is 1e78 and panics before FullMath can handle the product.
        Round6USDC bigUsdc = new Round6USDC();
        LendingMarket big = new LendingMarket(IERC20(address(bigUsdc)), 0, HYPE_PERP, 62, 0, 0, 0);
        _quote(1);

        uint256 liquidity = 2e44;
        bigUsdc.mint(LENDER, liquidity);
        vm.startPrank(LENDER);
        bigUsdc.approve(address(big), type(uint256).max);
        big.supply(liquidity);
        vm.stopPrank();

        vm.deal(BORROWER, 1);
        vm.prank(BORROWER);
        big.depositCollateral{value: 1}();
        vm.prank(BORROWER);
        big.borrow(6e43); // 60% of the 1e44-base-unit value of one wei at 1e74

        skip(24 * 365 days); // 5% base rate grows the debt beyond the short-path boundary
        uint256 debt = big.debtOf(BORROWER);
        uint256 ltv = big.ltvBps(BORROWER, 1e74);
        assertGt(ltv, 19_000);
        emit log_named_uint("accrued debt (USDC base units)", debt);
        emit log_named_uint("LTV before overflowing liquidation (bps)", ltv);
        vm.expectRevert(abi.encodeWithSignature("Panic(uint256)", 0x11));
        big.liquidate(BORROWER, type(uint256).max);
    }
}
